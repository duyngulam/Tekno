# Entity Framework Migrations Script
# Run these commands from the Tekno.Infrastructure directory

# Create a new migration
dotnet ef migrations add MigrationName --startup-project ..\Tekno.Api

# Drop database (use with caution - all data will be lost!)
dotnet ef database drop --startup-project ..\Tekno.Api -f

# Update database to latest migration
dotnet ef database update --startup-project ..\Tekno.Api

# Update to specific migration
# dotnet ef database update MigrationName --startup-project ..\Tekno.Api

# Rollback to previous migration
# dotnet ef database update PreviousMigrationName --startup-project ..\Tekno.Api

# Remove last migration (if not applied to database)
# dotnet ef migrations remove --startup-project ..\Tekno.Api --force

# List all migrations
# dotnet ef migrations list --startup-project ..\Tekno.Api

# Generate SQL script for migration
# dotnet ef migrations script --startup-project ..\Tekno.Api --output migration.sql
